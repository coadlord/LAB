Experiment 1(B): Doubly Linked List

Python Program:

class Node:
    def __init__(self, data):
        self.data = data
        self.prev = None
        self.next = None

head = Node(10)
second = Node(20)
third = Node(30)

head.next = second
second.prev = head
second.next = third
third.prev = second

# Insertion
new_node = Node(40)
third.next = new_node
new_node.prev = third

# Display
temp = head
while temp:
    print(temp.data, end=" ")
    temp = temp.next

# Delete first node
head = head.next
head.prev = None

print("\nAfter deletion:")
temp = head
while temp:
    print(temp.data, end=" ")
    temp = temp.next

Output:
10 20 30 40
After deletion:
20 30 40
